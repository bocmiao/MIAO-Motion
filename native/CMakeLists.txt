cmake_minimum_required(VERSION 3.20)
project(MiaoCamera LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB BASE_SOURCES CONFIGURE_DEPENDS softcam/src/baseclasses/*.cpp)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS softcam/src/softcamcore/*.cpp)
add_library(softcam SHARED ${BASE_SOURCES} ${CORE_SOURCES} softcam/src/softcam/softcam.cpp softcam/src/softcam/softcam.def)
target_include_directories(softcam PRIVATE softcam/src softcam/src/baseclasses)
target_compile_definitions(softcam PRIVATE UNICODE _UNICODE WIN32 _CRT_SECURE_NO_WARNINGS)
target_compile_options(softcam PRIVATE /utf-8)
target_link_libraries(softcam PRIVATE strmiids winmm ole32 oleaut32 uuid advapi32 user32 gdi32)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/camera-payload.rc" CONTENT "101 RCDATA \"$<TARGET_FILE:softcam>\"\n")
add_executable(camera-register WIN32 register.cpp "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/camera-payload.rc")
add_dependencies(camera-register softcam)
target_compile_options(camera-register PRIVATE /utf-8)
target_link_libraries(camera-register PRIVATE shell32 ole32 advapi32 uuid)
target_link_options(camera-register PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(camera-sender smoke-sender.cpp)
target_include_directories(camera-sender PRIVATE softcam/src)
target_link_libraries(camera-sender PRIVATE softcam)
if(DEFINED ENV{GITHUB_ACTIONS})
  add_executable(desktop-file-drop ../scripts/native-file-drop.cpp)
  target_compile_options(desktop-file-drop PRIVATE /utf-8)
  target_link_libraries(desktop-file-drop PRIVATE shell32 ole32 user32 uuid)
endif()
